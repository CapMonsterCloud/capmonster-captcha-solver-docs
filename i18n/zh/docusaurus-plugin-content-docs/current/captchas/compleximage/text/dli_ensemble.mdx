## 如何获取 Base64

页面中的图片可以以 URL 形式存在，也可以已经被编码为 Base64 格式。要找到所需的值，请在验证码图片上右键，选择 **检查（Inspect）**，然后仔细查看 **Elements（元素）** 面板或网络请求 —— 在那里你可以找到图片 URL 或已编码的内容。

1. 在浏览器中打开显示验证码的网站。  
2. 右键点击验证码元素并选择 **检查（Inspect）**。

import asyncio

API_KEY = "YOUR_API_KEY"  # 指定您的 CapMonster Cloud API 密钥


async def solve_dli():

        options=ClientOptions(api_key=API_KEY)
    )

    # RecognitionComplexImageTaskRequest 不需要代理
    dli_request = RecognitionComplexImageTaskRequest(
        imagesBase64=[
            "your_image_base64",
        ],
        metadata={
            "Task": "dli",
        },
    )

    # 如有需要，可以检查余额
    balance = await client.get_balance()
    print("Balance:", balance)

    result = await client.solve_captcha(dli_request)
    print("Solution:", result)


asyncio.run(solve_dli())

// https://github.com/CapMonsterCloud/capmonster-dotnet-captcha-solver

using System;
using System.Threading.Tasks;
using Newtonsoft.Json;
using Zennolab.CapMonsterCloud;
using Zennolab.CapMonsterCloud.Requests;

class Program
{
    static async Task Main(string[] args)
    {
        const string API_KEY = "YOUR_API_KEY"; // 指定您的 CapMonster Cloud API 密钥

        var clientOptions = new ClientOptions
        {
            ClientKey = API_KEY
        };

        var cmCloudClient = CapMonsterCloudClientFactory.Create(clientOptions);

        // RecognitionComplexImageTaskRequest 不需要代理
        var dliRequest = new RecognitionComplexImageTaskRequest
        {
            ImagesBase64 = new[]
            {
                "your_image_base64"
            },

            Metadata = new RecognitionComplexImageTaskRequest.RecognitionMetadata
            {
                Task = "dli"
            }
        };

        // 如有需要，可以检查余额
        var balance = await cmCloudClient.GetBalanceAsync();
        Console.WriteLine("Balance: " + balance);

        var result = await cmCloudClient.SolveAsync(dliRequest);

        Console.WriteLine("Solution:");
        Console.WriteLine(
            JsonConvert.SerializeObject(
                result.Solution,
                Formatting.Indented
            )
        );
    }
}
